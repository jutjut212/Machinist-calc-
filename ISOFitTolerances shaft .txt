    public static Double[] getTolerance(double size, String toleranceClass) {
        String sizeRange = getSizeRange(size);
        if (sizeRange == null) {
            return null; // Or throw an exception if you prefer
        }

        Map<String, Double[]> tolerances = toleranceData.get(sizeRange);
        if (tolerances == null) {
            return null; // Or throw an exception
        }

        return tolerances.get(toleranceClass);
    }
