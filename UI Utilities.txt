
import java.awt.*;
import javax.swing.*;
import javax.swing.border.AbstractBorder;
import javax.swing.border.LineBorder;

public class UIUtils {

    private static Font labelFont = new Font("Arial", Font.PLAIN, 14);
    private static Font headerFont = new Font("Arial", Font.BOLD, 16);
    private static Font resultFont = new Font("Arial", Font.BOLD, 14);
    private static Color backgroundColor = Color.LIGHT_GRAY;
    private static Color foregroundColor = Color.BLACK;
    private static Color buttonColor = new Color(59, 89, 152);
    private static Color buttonTextColor = Color.WHITE;

    public static void applyDarkMode(Component component, boolean darkMode) {
        if (darkMode) {
            component.setBackground(Color.DARK_GRAY);
            component.setForeground(Color.WHITE);
            backgroundColor = Color.DARK_GRAY;
            foregroundColor = Color.WHITE;
            buttonColor = new Color(100, 100, 100);
        } else {
            component.setBackground(Color.LIGHT_GRAY);
            component.setForeground(Color.BLACK);
            backgroundColor = Color.LIGHT_GRAY;
            foregroundColor = Color.BLACK;
            buttonColor = new Color(59, 89, 152);
        }
        if (component instanceof Container) {
            for (Component child : ((Container) component).getComponents()) {
                applyDarkMode(child, darkMode);
            }
        }
    }

    public static JLabel createLabel(String text) {
        JLabel label = new JLabel(text);
        label.setFont(labelFont);
        label.setForeground(foregroundColor);
        return label;
    }

    public static JLabel createHeaderLabel(String text) {
        JLabel label = new JLabel(text);
        label.setFont(headerFont);
        label.setForeground(foregroundColor);
        return label;
    }

    public static JLabel createResultLabel(String text) {
        JLabel label = new JLabel(text);
        label.setFont(resultFont);
        label.setForeground(foregroundColor);
        return label;
    }

    public static JTextField createTextField(String placeholder) {
        JTextField textField = new JTextField();
        textField.setForeground(foregroundColor);
        textField.setBorder(new LineBorder(Color.GRAY));
        textField.setCaretColor(foregroundColor);
        textField.putClientProperty("JTextField.placeholderText", placeholder);
        return textField;
    }

    public static JButton createRoundedButton(String text) {
        JButton button = new JButton(text);
        button.setBackground(buttonColor);
        button.setForeground(buttonTextColor);
        button.setFocusPainted(false);
        button.setBorder(BorderFactory.createEmptyBorder(5, 15, 5, 15));
        button.setOpaque(true);
        button.setBorder(new RoundedBorder(10));
        return button;
    }

    static class RoundedBorder extends AbstractBorder {

        private int radius;

        public RoundedBorder(int radius) {
            this.radius = radius;
        }

        @Override
        public void paintBorder(Component c, Graphics g, int x, int y, int width, int height) {
            Graphics2D g2d = (Graphics2D) g.create();
            g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2d.setColor(buttonColor);
            g2d.drawRoundRect(x, y, width - 1, height - 1, radius, radius);
            g2d.dispose();
        }

        @Override
        public Insets getBorderInsets(Component c) {
            return new Insets(this.radius + 1, this.radius + 1, this.radius + 2, this.radius);
        }

        @Override
        public Insets getBorderInsets(Component c, Insets insets) {
            insets.left = insets.right = insets.top = insets.bottom = this.radius;
            return insets;
        }
    }
}
