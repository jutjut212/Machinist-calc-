    public static Tolerance getTolerance(double size, String toleranceClass) {
        String sizeRange = getSizeRange(size);
        if (sizeRange == null) {
            return null; // Or throw an exception if you prefer
        }

        Map<String, Double[]> tolerances = toleranceData.get(sizeRange);
        if (tolerances == null) {
            return null; // Or throw an exception
        }

        Double[] deviations = tolerances.get(toleranceClass);
        if (deviations == null) {
            return null;
        }
        return new Tolerance(deviations[0], deviations[1]);
    }
